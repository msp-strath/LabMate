\begin{thm}
  If $\hasType{\Gamma}{t}{T}$ with $t$ of syntactic category $\Chk$, then there exists a term $t'$ such that $\chkrel{t}{t'}$ and $\ctxValid{\Gamma}\checkType{t'}{T}$.

  If $\hasType{\Gamma}{t}{T}$ with $t$ of syntactic category $\Syn$, then there exist $e$, $S$ such that $\synrel{t}{e}$, $\ctxValid{\Gamma}\synthType{e}{S}$ and $\ctxType{\Gamma}{S \equiv T}$.
\end{thm}
