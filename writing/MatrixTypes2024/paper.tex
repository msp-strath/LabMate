\begin{gather*}
  \fbox{\Type T}
  \\ \\
  \infer{\Type \typeIntro{\Pi}{S \bsl x . T}}{\Type S & x \in S \vdash \Type T}
  \qquad\qquad
  \infer{\Type \typeIntro{\Sigma}{S \bsl x . T}}{\Type S & x \in S \vdash \Type T}
  \\ \\
  \infer{\Type \typeIntro{\Zero}{}}{}
  \qquad\qquad
  \infer{\Type \typeIntro{\One}{}}{}
  \\ \\
  \infer{\Type \typeIntro{\List}{T}}{\Type T}
  \\ \\
  \infer{\Type \typeIntro{\Matrix}{R\, C\, \bsl r. \bsl c. T\, rs\, cs}}{
    \Type R & \Type C & r \in R, c \in C \vdash \Type T
    & \typeIntro{\List}{R} \ni rs & \typeIntro{\List}{C} \ni cs
  }
\end{gather*}

\begin{gather*}
  \fbox{\{\Type T\} T $\in$ t}
  \\ \\
  \infer{\typeIntro{\Pi}{S \bsl x . T} \ni \bsl x. t}{x \in S \vdash T \ni t}
  \qquad\qquad
  \infer{\typeIntro{\Sigma}{S \bsl x . T} \ni \Pair{s}{t}}{S \ni s & \subst{T}{s:S} \ni t}
  \\ \\
  \infer{\One \ni \Nil}{}
  \\ \\
  \infer{\typeIntro{List}{T} \ni \Nil}{} \qquad\qquad
  \infer{\typeIntro{List}{T} \ni \intro{\one}{t}}{T \ni t}
  \qquad\qquad \infer{\typeIntro{List}{T} \ni \intro{\jux}{ts_{0}\;
      ts_{1}}}{\typeIntro{List}{T} \ni ts_{0} & \typeIntro{List}{T} \ni
    ts_{1}}
  \\ \\
  \infer{T \ni (e)}{e \in S & S \equiv T}
\end{gather*}

\begin{gather*}
  \fbox{ e $\in$ S \{\Type T\}}
  \\ \\
  \infer{t : T \in T}{\Type T & T \ni t}
  \\ \\
  \infer{f s \in \subst{T}{s : S}}{f \in \typeIntro{\Pi}{S \bsl x . T} & S \ni s}
  \\ \\
  \infer{e \elim{\fst}{} \in S}{e \in \typeIntro{\Sigma}{S \bsl x. T}}
  \qquad\qquad
  \infer{e \elim{\snd}{} \in \subst{T}{e \elim{\fst}{} }}{e \in \typeIntro{\Sigma}{S \bsl x. T}}
  \\ \\
  \infer{e T \in T}{e \in \typeIntro{\Zero}{} & \Type T}
\end{gather*}

Main points:
\begin{itemize}
\item operability of TS (admits non-magical presentation)
\item get the eq. theory of lists as a monoid (and canonical form of lists)
\item get a canonical form for matrices constructed by pasting
\end{itemize}
Objective: admissibality of $\infer{\Type T}{t : T}$\\
