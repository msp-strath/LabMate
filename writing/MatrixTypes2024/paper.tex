\begin{gather*}
  \fbox{\Type T}
  \\ \\
  \infer{\Type \typeIntro{\Pi}{S \bsl x . T}}{\Type S & x \in S \vdash \Type T}
  \qquad\qquad
  \infer{\Type \typeIntro{\Sigma}{S \bsl x . T}}{\Type S & x \in S \vdash \Type T}
  \\ \\
  \infer{\Type \typeIntro{\Zero}{}}{}
  \qquad\qquad
  \infer{\Type \One}{}
  \\ \\
  \infer{\Type \List{T}}{\Type T}
  \\ \\
  \infer{\Type \MRCT{\V{rs}}{\V{cs}}}{
    \Type R & \Type C & r \in R, c \in C \vdash \Type T
    & \List{R} \ni rs & \List{C} \ni cs
  }
\end{gather*}


\begin{gather*}
  \fbox{\{\Type T\} T $\in$ t}
  \\ \\
  \infer{\typeIntro{\Pi}{S \bsl x . T} \ni \bsl x. t}{x \in S \vdash T \ni t}
  \qquad\qquad
  \infer{\typeIntro{\Sigma}{S \bsl x . T} \ni \Pair{s}{t}}{S \ni s & \subst{T}{s:S} \ni t}
  \\ \\
  \infer{\One \ni \Nil}{}
  \\ \\
  \infer{\typeIntro{List}{T} \ni \Nil}{} \qquad\qquad
  \infer{\typeIntro{List}{T} \ni \one{t}}{T \ni t}
  \qquad\qquad \infer{\List{T} \ni \jux{\V{ts_{0}}}{\V{ts_{1}}}}{\typeIntro{List}{T} \ni ts_{0} & \typeIntro{List}{T} \ni  ts_{1}}
  \\ \\
  \infer{T \ni (e)}{e \in S & S \equiv T}
\end{gather*}

  \\ \\
  \infer{e\; T \in T}{e \in \typeIntro{\Zero}{} & \Type T}
\end{gather*}

\begin{gather*}
  \fbox{$\ctxType{\Gamma}{T} \quad \{{\ctxValid{\Gamma}}\}$}
  \\ \\
  \infer{\ctxType{\Gamma}{T}}{\ctxValid{\Gamma}}
  \\ \\
  \infer{\ctxType{\Gamma}{\typeIntro{\Pi}{S \gp\la{x} T}}}{\ctxType{\Gamma \cs x \hb S}{T}}
  \qquad\qquad
  \infer{\ctxType{\Gamma}{\typeIntro{\Sigma}{S \gp\la{x} T}}}{\ctxType{\Gamma \cs x \hb S}{T}}
  \\ \\
  \infer{\ctxType{\Gamma}{\typeIntro{\Zero}{}}}{\ctxValid{\Gamma}}
  \qquad\qquad
  \infer{\ctxType{\Gamma}{\One}}{\ctxValid{\Gamma}}
  \\ \\
  \infer{\ctxType{\Gamma}{\List{T}}}{\ctxType{\Gamma}{T}}
  \\ \\
  \infer{\ctxType{\Gamma}{\MRCT{\V{rs}}{\V{cs}}}}{
    \ctxType{\Gamma \cs r \hb  R \cs c \hb C}{T}
    & \hasType{\Gamma}{\V{rs}}{\List{R}} &\hasType{\Gamma}{\V{cs}}{\List{C}}
  }
\end{gather*}

\begin{gather*}
  \fbox{$\hasType{\Gamma}{t}{T} \quad \{ \ctxType{\Gamma}{T}  \}$}
  \\ \\
  \infer{\hasType{\Gamma}{\la{x} t}{\typeIntro{\Pi}{S \la{x}  T}}}{\hasType{\Gamma \cs x \hb S}{t}{T}}
  \qquad\qquad
  \infer{\hasType{\Gamma}{\Pair{s}{t}}{\typeIntro{\Sigma}{S \la{x} T}}}{\hasType{\Gamma}{s}{S} & \ctxType{\Gamma \cs x \hb S}{T} & \hasType{\Gamma}{t}{\substfor{T}{s}{x}}}
  \\ \\
  \infer{\hasType{\Gamma}{\Nil}{\One}}{\ctxValid{\Gamma}}
  \\ \\
  \infer{\hasType{\Gamma}{\Nil}{\List{T}}}{\ctxType{\Gamma}{T}}
  \qquad\qquad
  \infer{\hasType{\Gamma}{\one{t}}{\List{T}}}{\hasType{\Gamma}{t}{T}}
  \qquad\qquad \infer{\hasType{\Gamma}{\jux{\V{ts_{0}}}{\V{ts_{1}}}}{\List{T}}}{\hasType{\Gamma}{\V{ts_{0}}}{\List{T}} & \hasType{\Gamma}{\V{ts_{1}}}{\List{T}}}
  \\ \\
  \infer{\hasType{\Gamma}{\vnil}{\MRCT{\Nil}{\V{cs}}}}{\hasType{\Gamma}{\V{cs}}{\List{C}} & \ctxType{\Gamma \cs r \hb  R \cs c \hb C}{T}}
  \\ \\
  \infer{\hasType{\Gamma}{\hnil}{\MRCT{\V{rs}}{\Nil}}}{\hasType{\Gamma}{\V{rs}}{\List{R}} & \ctxType{\Gamma \cs r \hb  R \cs c \hb C}{T}}
  \\ \\
  \infer{\hasType{\Gamma}{\one{t}}{\MRCT{\one{r'}}{\one{c'}}}}{\hasType{\Gamma}{r'}{R} & \hasType{\Gamma}{c'}{C} & \ctxType{\Gamma \cs r \hb  R \cs c \hb C}{T} & \hasType{\Gamma}{t}{\substfor{\substfor{T}{r'}{r}}{c'}{c}}}
  \\ \\
  \infer{\hasType{\Gamma}{\vjux{\V{m_{0}}}{\V{m_{1}}}}{\MRCT{\jux{\V{rs_{0}}}{\V{rs_{1}}}}{\V{cs}}}}{\hasType{\Gamma}{\V{m_{0}}}{\MRCT{\V{rs_{0}}}{\V{cs}}} & \hasType{\Gamma}{\V{m_{1}}}{\MRCT{\V{rs_{1}}}{\V{cs}}}}
  \\ \\
  \infer{\hasType{\Gamma}{\hjux{\V{m_{0}}}{\V{m_{1}}}}{\MRCT{\V{rs}}{\jux{\V{cs_{0}}}{\V{cs_{1}}}}}}{\hasType{\Gamma}{\V{m_{0}}}{\MRCT{\V{rs}}{\V{cs_{0}}}} & \hasType{\Gamma}{\V{m_{1}}}{\MRCT{\V{rs}}{\V{cs_{1}}}}}
\end{gather*}

Main points:
\begin{itemize}
\item operability of TS (admits non-magical presentation)
\item get the eq. theory of lists as a monoid (and canonical form of lists)
\item get a canonical form for matrices constructed by pasting
\end{itemize}
Objective: admissibality of $\infer{\Type T}{t : T}$\\
Non-objective: alogirthm for type checking
